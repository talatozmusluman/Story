import os
import sqlite3
from PyPDF2 import PdfReader

class PDFLibrary:
    def __init__(self, db_path="pdf_index.db"):
        self.db_path = db_path
        self.conn = sqlite3.connect(self.db_path)
        self._create_table()

    # --------------------------
    # 1️⃣ Create table with require_update flag
    # --------------------------
    def _create_table(self):
        self.conn.execute("""
        CREATE VIRTUAL TABLE IF NOT EXISTS pdfs
        USING fts5(
            filename,
            content,
            platform,
            machine,
            module,
            tokenize='trigram'
        )
        """)
        # Regular table to track update flags (can't add columns to FTS5 directly)
        self.conn.execute("""
        CREATE TABLE IF NOT EXISTS pdf_flags (
            filename TEXT PRIMARY KEY,
            platform TEXT,
            machine TEXT,
            module TEXT,
            require_update INTEGER DEFAULT 1
        )
        """)
        self.conn.commit()

    # --------------------------
    # 2️⃣ Extract text from PDF
    # --------------------------
    @staticmethod
    def _extract_text(pdf_path):
        reader = PdfReader(pdf_path)
        text = ""
        for page in reader.pages:
            page_text = page.extract_text()
            if page_text:
                text += page_text + " "
        return text

    # --------------------------
    # 3️⃣ Mark a PDF as requiring update
    # --------------------------
    def mark_pdf_for_update(self, filename, platform, machine, module):
        self.conn.execute("""
            INSERT OR REPLACE INTO pdf_flags (filename, platform, machine, module, require_update)
            VALUES (?, ?, ?, ?, 1)
        """, (filename, platform, machine, module))
        self.conn.commit()

    # --------------------------
    # 4️⃣ Index all PDFs recursively
    # --------------------------
    def index_pdfs(self, root_folder):
        for platform_name in os.listdir(root_folder):
            platform_path = os.path.join(root_folder, platform_name)
            if not os.path.isdir(platform_path):
                continue

            for machine_name in os.listdir(platform_path):
                machine_path = os.path.join(platform_path, machine_name)
                if not os.path.isdir(machine_path):
                    continue

                for module_name in os.listdir(machine_path):
                    module_path = os.path.join(machine_path, module_name)
                    if not os.path.isdir(module_path):
                        continue

                    for pdf_file in os.listdir(module_path):
                        if not pdf_file.lower().endswith(".pdf"):
                            continue
                        pdf_path = os.path.join(module_path, pdf_file)
                        # mark new PDFs for update
                        self.mark_pdf_for_update(pdf_file, platform_name, machine_name, module_name)

        # Index only PDFs marked for update
        self.index_updated_pdfs(root_folder)

    # --------------------------
    # 5️⃣ Index only PDFs with require_update = 1
    # --------------------------
    def index_updated_pdfs(self, root_folder):
        cursor = self.conn.execute("SELECT filename, platform, machine, module FROM pdf_flags WHERE require_update = 1")
        updated_files = cursor.fetchall()

        for filename, platform, machine, module in updated_files:
            pdf_path = os.path.join(root_folder, platform, machine, module, filename)
            if os.path.exists(pdf_path):
                text = self._extract_text(pdf_path)
                if text.strip():
                    # insert or replace into FTS5 table
                    self.conn.execute(
                        "INSERT OR REPLACE INTO pdfs (filename, content, platform, machine, module) VALUES (?, ?, ?, ?, ?)",
                        (filename, text, platform, machine, module)
                    )
                    # reset require_update flag
                    self.conn.execute("""
                        UPDATE pdf_flags SET require_update = 0
                        WHERE filename = ? AND platform = ? AND machine = ? AND module = ?
                    """, (filename, platform, machine, module))
        self.conn.commit()

    # --------------------------
    # 6️⃣ Search PDFs with optional hierarchy filter
    # --------------------------
    def search(self, keyword, platform=None, machine=None, module=None):
        query = "SELECT filename, platform, machine, module FROM pdfs WHERE content MATCH ?"
        params = [keyword]

        if platform:
            query += " AND platform = ?"
            params.append(platform)
        if machine:
            query += " AND machine = ?"
            params.append(machine)
        if module:
            query += " AND module = ?"
            params.append(module)

        cursor = self.conn.execute(query, params)
        return [(row[0], row[1], row[2], row[3]) for row in cursor]

    # --------------------------
    # 7️⃣ Close DB
    # --------------------------
    def close(self):
        self.conn.close()


if __name__ == "__main__":
    pdf_lib = PDFLibrary()

    # Index PDFs (only new/updated ones will be processed)
    pdf_lib.index_pdfs("pdf_root")

    # User search
    keyword = input("Enter keyword: ").strip()
    platform = input("Platform (optional): ").strip() or None
    machine = input("Machine (optional): ").strip() or None
    module = input("Module (optional): ").strip() or None

    results = pdf_lib.search(keyword, platform, machine, module)
    for filename, platform_name, machine_name, module_name in results:
        print(f"{platform_name}/{machine_name}/{module_name}/{filename}")

    pdf_lib.close()
